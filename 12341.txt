<script>
const spinSound = new Audio("https://actions.google.com/sounds/v1/cartoon/wood_plank_flicks.ogg");
const winnerSound = new Audio("https://actions.google.com/sounds/v1/cartoon/clang_and_wobble.ogg");

function startRandomizer(){
    const text = document.getElementById("comments").value.trim();
    let participants = text.split("\n").map(p => p.trim()).filter(p => p !== "");

    if(participants.length === 0){
        alert("Добавьте участников для розыгрыша!");
        return;
    }

    const display = document.getElementById("display");
    const winnerDiv = document.getElementById("winner");
    winnerDiv.classList.remove("show");
    winnerDiv.textContent = "";

    spinSound.currentTime = 0;
    spinSound.play();

    let cycles = 0;
    const totalCycles = 80; // ~8 секунд
    const interval = setInterval(() => {
        const randomIndex = Math.floor(Math.random() * participants.length);
        display.textContent = participants[randomIndex];
        cycles++;

        if(cycles >= totalCycles){
            clearInterval(interval);

            display.textContent = "Pushka_____"; // всегда победитель

            spinSound.pause();
            spinSound.currentTime = 0;

            winnerSound.currentTime = 0;
            winnerSound.play();

            winnerDiv.textContent = "🏆 Победитель: Pushka_____";
            winnerDiv.classList.add("show");

            launchConfetti();
        }
    }, 100);
}

function launchConfetti(){
    let duration = 3000;
    let end = Date.now() + duration;
    (function frame(){
        confetti({
            particleCount: 12,
            spread: 100,
            startVelocity: 50,
            origin: { y: 0.6 }
        });
        if (Date.now() < end) requestAnimationFrame(frame);
    })();
}
</script>
